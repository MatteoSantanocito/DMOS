# ══════════════════════════════════════════════════════════════════════════
#
# Prerequisiti:
#   - I 3 cluster K3s sono attivi e raggiungibili
#   - Nginx Ingress Controller deployato su ogni cluster (porta 30080)
#   - Online Boutique è deployato su tutti i cluster
#   - locustfile_multiingress.py è nella cartella experiments/
#
# IMPORTANTE: usa locustfile_multiingress.py (NON locustfile_scenarios.py)
#   - locustfile_multiingress.py: 3 UserClass separate, una per cluster,
#     hardcoded su :30080 (Nginx Ingress). Traffico reale multi-cluster.
#   - locustfile_scenarios.py: single-ingress (usa --host), manda tutto
#     su cluster1 soltanto. Incompatibile con setup Nginx attuale.
#
# Tra un test e l'altro:
#   1. Locust si ferma da solo a fine scenario
#   2. Aspetta che il collector finisca (ha ~5 min di margine)
#   3. Stoppa DMOS (Ctrl+C) e riavvialo per pulire lo stato del predittore
#   4. Opzionale: resetta le repliche a 1 per cluster prima del test
#      kubectl scale deployment frontend -n online-boutique --replicas=1 \
#        --context cluster1 --context cluster2 --context cluster3
#
# ══════════════════════════════════════════════════════════════════════════

# │  CAPACITY TEST (una tantum — calibra services.yaml)                 │
# └──────────────────────────────────────────────────────────────────────┘
#
# Scopo: trovare il knee point rps/replica del frontend e aggiornare
#        capacity_req_per_sec in config/services.yaml.
# Da eseguire solo quando cambia l'hardware o dopo modifiche al frontend.
# Durata: ~15-35 min (dipende da quanti step prima della violazione SLA)

# Step 1: ferma DMOS e scala frontend a 1 replica/cluster
kubectl scale deployment frontend -n online-boutique --replicas=1 --context cluster1
kubectl scale deployment frontend -n online-boutique --replicas=1 --context cluster2
kubectl scale deployment frontend -n online-boutique --replicas=1 --context cluster3

# Step 2: esegui il capacity test headless
locust -f experiments/locustfile_capacity.py --host http://ignored --headless

# Step 3: leggi l'output console — cerca la riga:
#   → Aggiorna config/services.yaml:
#       frontend:
#         capacity_req_per_sec: XX
#   → Aggiorna analyze_test_complete.py:
#       SERVICE_CAPACITY['frontend'] = YY

# Step 4: se il valore raccomandato è diverso dall'attuale (30 rps),
#         aggiorna config/services.yaml e analyze_test_complete.py


# ══════════════════════════════════════════════════════════════════════════
# SCENARI DI TEST DMOS (usa locustfile_multiingress.py)
# ══════════════════════════════════════════════════════════════════════════
#
# locustfile_multiingress.py NON usa --host: i cluster IP sono hardcoded
# nel file (192.168.1.245:30080, .246:30080, .247:30080)


# ┌──────────────────────────────────────────────────────────────────────┐
# │  SCENARIO 1: flash_crowd (~20 min) — Spike improvviso               │
# └──────────────────────────────────────────────────────────────────────┘
# Fasi: warm-up(3') → flash-spike(4') → sustained-peak(5') →
#        partial-decline(3') → full-decline(3') → cooldown(2') = 20 min

# Terminal 1: DMOS
python src/dmos_main.py

# Terminal 2: Collector (20 min scenario + 5 min margine = 25 min)
python experiments/collect_metrics_simple.py 25 -s flash_crowd

# Terminal 3: Locust (si ferma da solo dopo 20 min)
$env:DMOS_SCENARIO="flash_crowd"
locust -f experiments/locustfile_multiingress.py --headless

# Terminal 4: Analisi (dopo che collector finisce)
python experiments/analyze_test_complete.py results/<HHMMSS>_<YYYYMMDD>_flash_crowd.jsonl


# ┌──────────────────────────────────────────────────────────────────────┐
# │  SCENARIO 2: double_wave (~26 min) — Doppia onda                   │
# └──────────────────────────────────────────────────────────────────────┘
# Fasi: warm-up(2') → wave1-ramp(3') → wave1-peak(4') → valley-descent(3') →
#        valley(2') → wave2-ramp(3') → wave2-peak(4') → final-decline(3') →
#        cooldown(2') = 26 min

# Terminal 1: DMOS
python src/dmos_main.py

# Terminal 2: Collector (26 + 5 = 31 min)
python experiments/collect_metrics_simple.py 31 -s double_wave

# Terminal 3: Locust
$env:DMOS_SCENARIO="double_wave"
locust -f experiments/locustfile_multiingress.py --headless

# Analisi:
python experiments/analyze_test_complete.py results/<HHMMSS>_<YYYYMMDD>_double_wave.jsonl


# ┌──────────────────────────────────────────────────────────────────────┐
# │  SCENARIO 3: sinusoidal (~34 min) — Oscillante anti-flapping       │
# └──────────────────────────────────────────────────────────────────────┘
# Fasi: warm-up(2') → sinusoidal 80-300 utenti period=6min (30') →
#        cooldown(2') = 34 min

# Terminal 1: DMOS
python src/dmos_main.py

# Terminal 2: Collector (34 + 5 = 39 min)
python experiments/collect_metrics_simple.py 39 -s sinusoidal

# Terminal 3: Locust
$env:DMOS_SCENARIO="sinusoidal"
locust -f experiments/locustfile_multiingress.py --headless

# Analisi:
python experiments/analyze_test_complete.py results/<HHMMSS>_<YYYYMMDD>_sinusoidal.jsonl
